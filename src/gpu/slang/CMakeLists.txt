function(read_slang_package bundle format)
  foreach(asset IN ITEMS plan.tsv SHA256SUMS)
    if(NOT EXISTS "${bundle}/${asset}")
      message(FATAL_ERROR "${bundle} doit contenir ${asset}.")
    endif()
  endforeach()
  file(STRINGS "${bundle}/plan.tsv" header LIMIT_COUNT 1)
  if(NOT header MATCHES "^${format}\t1920\t1080\t([0-9a-f]+)$")
    message(FATAL_ERROR "En-tête pour ${format} invalide dans ${bundle}.")
  endif()
  string(LENGTH "${CMAKE_MATCH_1}" model_length)
  if(NOT model_length EQUAL 64)
    message(FATAL_ERROR "Empreinte ONNX invalide dans ${bundle}.")
  endif()
  set(package_model "${CMAKE_MATCH_1}" PARENT_SCOPE)
  file(SHA256 "${bundle}/SHA256SUMS" digest)
  set(package_sha "${digest}" PARENT_SCOPE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${bundle}/SHA256SUMS")
endfunction()

# Le Resize expérimental utilise le même VkDevice et la même arène que les autres opérations du graphe.
option(KIRARI_SLANG_RESIZE4 "Essayer le Resize bilinéaire FP16 ×4 Slang sur la voie Vulkan" OFF)
if(KIRARI_SLANG_RESIZE4 AND NOT APPLE)
  find_program(KIRARI_SLANG_COMPILER slangc REQUIRED)
  find_program(KIRARI_SLANG_SPIRV_OPT spirv-opt REQUIRED)
  find_program(KIRARI_SLANG_SPIRV_VAL spirv-val REQUIRED)
  set(resize_spv "${CMAKE_CURRENT_BINARY_DIR}/resize4.spv")
  set(resize_raw "${CMAKE_CURRENT_BINARY_DIR}/resize4-raw.spv")
  set(resize_inc "${CMAKE_CURRENT_BINARY_DIR}/resize4.spv.inc")
  add_custom_command(OUTPUT "${resize_inc}" BYPRODUCTS "${resize_raw}" "${resize_spv}"
    COMMAND "${KIRARI_SLANG_COMPILER}" "${CMAKE_CURRENT_SOURCE_DIR}/resize4.slang" -entry main
            -target spirv -O3 -fp-mode precise -o "${resize_raw}"
    COMMAND "${KIRARI_SLANG_SPIRV_OPT}" --trim-capabilities "${resize_raw}" -o "${resize_spv}"
    COMMAND "${KIRARI_SLANG_SPIRV_VAL}" --target-env "${KIRARI_SPIRV_TARGET_ENV}" "${resize_spv}"
    COMMAND "${CMAKE_COMMAND}" "-DKIRARI_SPIRV_IN=${resize_spv}" "-DKIRARI_SPIRV_OUT=${resize_inc}"
            -P "${KIRARI_SPIRV_SCRIPT}"
    DEPENDS resize4.slang "${KIRARI_SPIRV_SCRIPT}" VERBATIM)
  target_sources(kirari_backend_slang PRIVATE resize4.cpp "${resize_inc}")
  target_include_directories(kirari_backend_slang PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(kirari_backend_slang PRIVATE KIRARI_SLANG_RESIZE4)
  add_executable(kirari_slang_resize4 EXCLUDE_FROM_ALL
    "${PROJECT_SOURCE_DIR}/tools/bench/kernel_portable/interpolation/resize4_slang.cpp"
    resize4.cpp "${resize_inc}")
  target_include_directories(kirari_slang_resize4 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(kirari_slang_resize4 PRIVATE kirari::vk_backend kirari::kernels)
  if(MSVC)
    target_compile_options(kirari_slang_resize4 PRIVATE /fp:strict)
  else()
    target_compile_options(kirari_slang_resize4 PRIVATE -ffp-contract=off)
  endif()
  kirari_set_warnings(kirari_slang_resize4)
endif()

set(slang_catalogue "")
set(slang_models "")
set(slang_folder "")
foreach(bundle IN ITEMS "${KIRARI_SLANG_BUNDLE}" ${KIRARI_SLANG_ADDITIONAL_BUNDLES})
  read_slang_package("${bundle}" "${slang_format}")
  if(package_model IN_LIST slang_models)
    message(FATAL_ERROR "Modèle Slang en double dans ${bundle}.")
  endif()
  if(slang_models)
    set(slang_folder "/${package_model}")
  endif()
  list(APPEND slang_models "${package_model}")
  string(APPEND slang_catalogue
    "  {\"slang${slang_folder}\", \"${package_model}\", \"${package_sha}\"},\n")
  set(slang_assets "${bundle}/plan.tsv" "${bundle}/SHA256SUMS")
  file(GLOB slang_data CONFIGURE_DEPENDS "${bundle}/*.spv" "${bundle}/*.metal" "${bundle}/*.bin")
  list(APPEND slang_assets ${slang_data})
  add_custom_command(TARGET kirari_backend_slang POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:kirari_backend_slang>/slang${slang_folder}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      ${slang_assets} "$<TARGET_FILE_DIR:kirari_backend_slang>/slang${slang_folder}/"
    VERBATIM)
  install(FILES ${slang_assets} DESTINATION "slang${slang_folder}" COMPONENT slang)
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/slang_packages.inc" CONTENT "${slang_catalogue}")
install(TARGETS kirari_backend_slang LIBRARY DESTINATION . COMPONENT slang RUNTIME DESTINATION . COMPONENT slang)
