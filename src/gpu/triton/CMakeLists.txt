set(triton_catalogue "")
function(kirari_triton_packages bundles arch platform)
  target_compile_definitions(kirari_backend_triton PRIVATE KIRARI_TRITON_${platform})
  set(index 0)
  set(models)
  foreach(bundle IN LISTS bundles)
    if(NOT EXISTS "${bundle}/plan.tsv" OR NOT EXISTS "${bundle}/SHA256SUMS")
      message(FATAL_ERROR "Le paquet Triton ${arch} doit contenir plan.tsv et SHA256SUMS (${bundle})")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${bundle}/SHA256SUMS" "${bundle}/plan.tsv")
    file(STRINGS "${bundle}/plan.tsv" header LIMIT_COUNT 1)
    if(header MATCHES "^(span-triton-v1)\t(1920)\t(1080)\t([0-9a-f]+)$")
      set(model_kind Span)
    elseif(header MATCHES "^(drba-triton-v1|drba-triton-motion-v1)\t(1920|1280)\t(1088|768)\t([0-9a-f]+)$")
      if((CMAKE_MATCH_2 STREQUAL "1920" AND NOT CMAKE_MATCH_3 STREQUAL "1088") OR
         (CMAKE_MATCH_2 STREQUAL "1280" AND NOT CMAKE_MATCH_3 STREQUAL "768"))
        message(FATAL_ERROR "Taille du paquet Triton DistilDRBA invalide dans ${bundle}")
      endif()
      if(CMAKE_MATCH_1 STREQUAL "drba-triton-motion-v1")
        set(model_kind DistilMotion)
      else()
        set(model_kind Distil)
      endif()
    elseif(header MATCHES "^(rife-triton-motion-v1)\t(1920)\t(1088)\t([0-9a-f]+)$")
      set(model_kind RifeMotion)
    elseif(header MATCHES "^(rife-lite-triton-motion-v1)\t(1920)\t(1088)\t([0-9a-f]+)$")
      set(model_kind RifeLiteMotion)
    else()
      message(FATAL_ERROR "Entête Triton invalide dans ${bundle}")
    endif()
    set(width "${CMAKE_MATCH_2}")
    set(height "${CMAKE_MATCH_3}")
    set(model "${CMAKE_MATCH_4}")
    string(LENGTH "${model}" model_length)
    if(NOT model_length EQUAL 64 OR model IN_LIST models)
      message(FATAL_ERROR "Empreinte de modèle Triton invalide ou en double pour ${arch} (${bundle})")
    endif()
    list(APPEND models "${model}")
    file(SHA256 "${bundle}/SHA256SUMS" package_sha)
    set(folder "${arch}")
    if(index GREATER 0)
      set(folder "${arch}-${model}")
    endif()
    string(APPEND triton_catalogue
      "  {\"${arch}\", \"${folder}\", \"${model}\", \"${package_sha}\", ModelKind::${model_kind}, ${width}, ${height}},\n")
    set(target kirari_triton_bundle_${platform}_${index})
    add_custom_target("${target}"
      COMMAND "${CMAKE_COMMAND}" "-Dsource=${bundle}"
        "-Ddestination=$<TARGET_FILE_DIR:kirari_backend_triton>/triton/${folder}"
        -P "${PROJECT_SOURCE_DIR}/cmake/StageLocalAsset.cmake"
      VERBATIM)
    add_dependencies(kirari_backend_triton "${target}")
    install(DIRECTORY "${bundle}/" DESTINATION "triton/${folder}" COMPONENT triton)
    math(EXPR index "${index} + 1")
  endforeach()
  set(triton_catalogue "${triton_catalogue}" PARENT_SCOPE)
endfunction()

if(KIRARI_TRITON_BUNDLE)
  if(NOT KIRARI_TRITON_CUDA_ARCH MATCHES "^sm[0-9]+$")
    message(FATAL_ERROR "Architecture Triton CUDA invalide : ${KIRARI_TRITON_CUDA_ARCH}")
  endif()
  if(KIRARI_TRITON_CUDA_INCLUDE_DIR)
    if(NOT EXISTS "${KIRARI_TRITON_CUDA_INCLUDE_DIR}/cuda.h")
      message(FATAL_ERROR "En-tête cuda.h absent de KIRARI_TRITON_CUDA_INCLUDE_DIR")
    endif()
    set(triton_cuda_headers "${KIRARI_TRITON_CUDA_INCLUDE_DIR}")
  else()
    find_package(CUDAToolkit REQUIRED)
    set(triton_cuda_headers ${CUDAToolkit_INCLUDE_DIRS})
  endif()
  target_sources(kirari_backend_triton PRIVATE cuda.cpp ../zml/cuda_timeline.cpp)
  target_link_libraries(kirari_backend_triton PRIVATE kirari::cuda_driver)
  target_include_directories(kirari_backend_triton SYSTEM PRIVATE ${triton_cuda_headers})
  if(KIRARI_TRITON_CUDA_GRAPHS)
    target_compile_definitions(kirari_backend_triton PRIVATE KIRARI_TRITON_CUDA_GRAPHS)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND KIRARI_TRITON_CUDA_TIMELINE)
    target_compile_definitions(kirari_backend_triton PRIVATE KIRARI_ZML_CUDA_TIMELINE)
  endif()
  target_compile_definitions(kirari_backend_triton PRIVATE TRITON_PACKAGE_CUDA_ARCH="${KIRARI_TRITON_CUDA_ARCH}")
  kirari_triton_packages("${KIRARI_TRITON_BUNDLE}" "${KIRARI_TRITON_CUDA_ARCH}" CUDA)
endif()
if(KIRARI_TRITON_HIP_BUNDLE)
  if(NOT KIRARI_TRITON_HIP_ARCH MATCHES "^gfx[0-9a-f]+$")
    message(FATAL_ERROR "Architecture Triton ROCm invalide : ${KIRARI_TRITON_HIP_ARCH}")
  endif()
  if(NOT KIRARI_TRITON_HIP_INCLUDE_DIR)
    find_path(triton_hip_sdk hip/hip_runtime_api.h HINTS "$ENV{HIP_PATH}/include" /opt/rocm/include REQUIRED)
    set(KIRARI_TRITON_HIP_INCLUDE_DIR "${triton_hip_sdk}")
  endif()
  if(NOT EXISTS "${KIRARI_TRITON_HIP_INCLUDE_DIR}/hip/hip_runtime_api.h")
    message(FATAL_ERROR "En-tête ROCm hip/hip_runtime_api.h absent de KIRARI_TRITON_HIP_INCLUDE_DIR")
  endif()
  set_property(SOURCE hip.cpp PROPERTY COMPILE_DEFINITIONS __HIP_PLATFORM_AMD__)
  target_sources(kirari_backend_triton PRIVATE hip.cpp)
  target_include_directories(kirari_backend_triton SYSTEM PRIVATE "${KIRARI_TRITON_HIP_INCLUDE_DIR}")
  target_compile_definitions(kirari_backend_triton PRIVATE TRITON_PACKAGE_HIP_ARCH="${KIRARI_TRITON_HIP_ARCH}")
  kirari_triton_packages("${KIRARI_TRITON_HIP_BUNDLE}" "${KIRARI_TRITON_HIP_ARCH}" HIP)
endif()

  target_include_directories(kirari_backend_triton SYSTEM PRIVATE "${KIRARI_TRITON_XPU_INCLUDE_DIR}")
  kirari_triton_packages("${KIRARI_TRITON_XPU_BUNDLE}" arc-a770 XPU)
endif()
