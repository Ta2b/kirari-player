function(kirari_zml_build_asset source relative)
  list(APPEND zml_stage_commands COMMAND "${CMAKE_COMMAND}" "-Dsource=${source}"

  set(zml_stage_commands "${zml_stage_commands}" PARENT_SCOPE)

# Le paquet amont complet garde ses bibliothèques, ses compilateurs et les licences qui les accompagnent.
# Le préfixe d'installation de ce composant est le dossier des modules du lecteur (PlugIns sur macOS).
if(KIRARI_ZML_BUNDLE)
  set(zml_platforms cuda rocm oneapi)
  if(APPLE)
    set(zml_platforms metal)
  endif()
  set(zml_present FALSE)
  foreach(platform IN LISTS zml_platforms)
    if(EXISTS "${KIRARI_ZML_BUNDLE}/${platform}")
      set(plugin "${CMAKE_SHARED_LIBRARY_PREFIX}zml_${platform}${CMAKE_SHARED_LIBRARY_SUFFIX}")
      if(NOT EXISTS "${KIRARI_ZML_BUNDLE}/${platform}/lib/${plugin}")
        message(FATAL_ERROR "Le paquet ZML ${platform} doit contenir lib/${plugin}")
      endif()
      if(platform STREQUAL "metal")
        foreach(tool metal air-as air-opt metallib)
          if(NOT EXISTS "${KIRARI_ZML_BUNDLE}/metal/toolchain/bin/${tool}")
            message(FATAL_ERROR "Le compilateur Metal du runtime ZML doit contenir toolchain/bin/${tool}")
          endif()
        endforeach()
      endif()
      if(platform STREQUAL "cuda")
        # Le bundle PJRT CUDA contient cuDNN, NVRTC, NCCL et les outils de NVIDIA. Le
        # destinataire peut le préparer et l'installer sur son poste. Aucun paquet de release
        # n'installe ce composant, même quand son build a aussi activé le module TensorRT.
        install(DIRECTORY "${KIRARI_ZML_BUNDLE}/cuda" DESTINATION zml USE_SOURCE_PERMISSIONS
                COMPONENT zml_cuda_local EXCLUDE_FROM_ALL)
      else()
        install(DIRECTORY "${KIRARI_ZML_BUNDLE}/${platform}" DESTINATION zml USE_SOURCE_PERMISSIONS COMPONENT zml)
      endif()
      kirari_zml_build_asset("${KIRARI_ZML_BUNDLE}/${platform}" "${platform}")
      set(zml_present TRUE)
    endif()
  endforeach()
  if(NOT zml_present)
    message(FATAL_ERROR "Aucun runtime ZML compatible avec le système cible dans ${KIRARI_ZML_BUNDLE}")
  endif()
endif()
